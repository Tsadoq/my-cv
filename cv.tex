% Bullet list used by \joborrole. Tight vertical metrics so it sits flush
% under the role header.
\newenvironment{rolebullets}{%
  \begin{itemize}[leftmargin=2ex, nosep, noitemsep, topsep=0pt, partopsep=0pt]%
    \setlength{\parskip}{0pt}%
    \renewcommand{\labelitemi}{\bullet}%
}{%
  \end{itemize}%
}

% \joborrole{<title>}{<organization>}{<location>}{<dates>}{<\item list>}
%
% Builds the header row(s) directly (mirroring \cventry's two-row layout)
% instead of delegating to \cventry. Reasons:
%   - \cventry puts the description inside a \multicolumn cell of a tabular*,
%     which doesn't measure block-level list content correctly and causes
%     entries to visually overlap.
%   - Calling \cventry with an empty description still leaves an empty
%     multicolumn row that adds a visible gap before the bullets.
% \needspace avoids orphaning a header at the bottom of a page, and
% \nopagebreak between header and bullets keeps them together.
\newcommand{\joborrole}[5]{%
  \needspace{5\baselineskip}%
  \setlength\tabcolsep{0pt}%
  \setlength{\extrarowheight}{-1pt}%
  \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}} L{\textwidth - 4.5cm} R{4.5cm}}
    \ifempty{#2#3}
      {\entrytitlestyle{#1} & \entrydatestyle{#4} \\}
      {\entrytitlestyle{#2} & \entrylocationstyle{#3} \\
       \entrypositionstyle{#1} & \entrydatestyle{#4} \\}
  \end{tabular*}%
  \nopagebreak[4]\par\vspace{-2mm}%
  \begin{rolebullets}
    #5
  \end{rolebullets}%
  \vspace{0mm}%
}
