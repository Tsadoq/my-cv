\cvsection{Notable Projects}

\begin{cventries}
\footnotesize
\begin{itemize}[leftmargin=*, nosep, label={}]
  \item \textbf{ErisForge: Behavior Control for LLMs} \hfill \href{https://github.com/Tsadoq/erisforge}{\color{awesome}github.com/Tsadoq/erisforge} \\
  Python library for modifying \textbf{Large Language Models (LLMs)} by ablating or inducing specific behaviors through internal direction manipulation. Compatible with almost any transformer-based LLM, avoiding reliance on transformer lenses or architecture-specific tools. Useful for \textbf{refusal behavior control}, tone/personality customization, censorship research, and adversarial testing. \textbf{200+ stars on GitHub}, 100 points on HackerNews.
  \vspace{2pt}

  \item \textbf{Haystack Documentation AI Agent} \hfill \href{https://github.com/Tsadoq/haystack-documentation-agent}{\color{awesome}github.com/Tsadoq/haystack-documentation-agent} \\
  Stateless GenAI-powered agent that browses the \textbf{Haystack} library documentation and writes code for the user; built on Haystack and Streamlit.
  \vspace{2pt}

  \item \textbf{KomPY: Python Wrapper for Komoot APIs} \hfill \href{https://github.com/Tsadoq/kompy}{\color{awesome}github.com/Tsadoq/kompy} \\
  Pythonic wrapper for \textbf{Komoot APIs} with typed datatypes and streamlined integration with the most used Python \textbf{geospatial libraries}.
  \vspace{2pt}

  \item \textbf{iCarl — Incremental Learning SOTA Improvement} \hfill {\color{awesome-skyblue} PoliTO, Graded 27/30} \\
  Re-implementation of the SOTA \textbf{iCarl} classification training strategy in PyTorch + Scikit-learn, improving results via \textbf{image clustering} and \textbf{averaging}.
\end{itemize}
